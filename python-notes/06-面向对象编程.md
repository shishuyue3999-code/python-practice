# 06-面向对象编程（OOP）

## 1. 类与对象

```python
class Dog:
    # 类属性（所有实例共享）
    species = "Canis familiaris"
    
    # 构造方法（初始化实例）
    def __init__(self, name, age):
        self.name = name    # 实例属性
        self.age = age
    
    # 实例方法
    def bark(self):
        return f"{self.name} says: 汪汪!"
    
    def get_info(self):
        return f"{self.name} 今年 {self.age} 岁"

# 创建对象
my_dog = Dog("Buddy", 3)
print(my_dog.bark())
print(my_dog.get_info())
```

## 2. 三大特性

### 封装

```python
class BankAccount:
    def __init__(self, balance):
        self.__balance = balance  # 双下划线：私有属性
    
    def deposit(self, amount):
        if amount > 0:
            self.__balance += amount
    
    def get_balance(self):
        return self.__balance

account = BankAccount(100)
account.deposit(50)
print(account.get_balance())  # 150
# print(account.__balance)    # ❌ 无法直接访问
```

### 继承

```python
class Animal:
    def __init__(self, name):
        self.name = name
    
    def speak(self):
        raise NotImplementedError("子类必须实现此方法")

class Cat(Animal):
    def __init__(self, name, color):
        super().__init__(name)  # 调用父类构造方法
        self.color = color
    
    def speak(self):  # 重写
        return f"{self.name} says: 喵喵!"

my_cat = Cat("Kitty", "white")
print(my_cat.speak())
```

### 多态

```python
def animal_sound(animal: Animal):
    print(animal.speak())

animal_sound(Dog("Buddy", 3))   # Buddy says: 汪汪!
animal_sound(Cat("Kitty", 2))   # Kitty says: 喵喵!
```

## 3. 魔术方法（Dunder Methods）

```python
class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def __str__(self):          # print() 时调用
        return f"Vector({self.x}, {self.y})"
    
    def __repr__(self):         # 交互式显示时调用
        return f"Vector({self.x}, {self.y})"
    
    def __add__(self, other):   # + 运算符
        return Vector(self.x + other.x, self.y + other.y)
    
    def __eq__(self, other):    # == 运算符
        return self.x == other.x and self.y == other.y

v1 = Vector(1, 2)
v2 = Vector(3, 4)
print(v1 + v2)  # Vector(4, 6)
```

## 4. 类方法与静态方法

```python
class MyClass:
    count = 0
    
    @classmethod
    def increment(cls):
        cls.count += 1
    
    @staticmethod
    def is_valid(x):
        return x > 0

MyClass.increment()
print(MyClass.is_valid(5))
```
