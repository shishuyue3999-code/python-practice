# 04-函数

## 1. 定义函数

```python
def greet(name):
    """函数的文档字符串（docstring）"""
    return f"你好，{name}!"

# 调用
msg = greet("Alice")
print(msg)
```

## 2. 参数类型

```python
# 位置参数
def add(a, b):
    return a + b

add(2, 3)  # 5

# 默认参数
def greet(name, greeting="你好"):
    return f"{greeting}，{name}!"

greet("Bob")           # "你好，Bob！"
greet("Bob", "早上好")  # "早上好，Bob！"

# 关键字参数（不受顺序限制）
add(b=5, a=3)

# 可变参数 *args（接收任意多个位置参数，组成元组）
def total(*numbers):
    return sum(numbers)

total(1, 2, 3, 4)  # 10

# 可变关键字参数 **kwargs（接收任意多个关键字参数，组成字典）
def info(**kwargs):
    for k, v in kwargs.items():
        print(f"{k}: {v}")

info(name="Alice", age=25)

# 组合使用（顺序：位置参数 > *args > 默认参数 > **kwargs）
def func(a, b, *args, c=10, **kwargs):
    pass
```

## 3. 返回值

```python
# 返回多个值（实际是返回元组）
def get_min_max(numbers):
    return min(numbers), max(numbers)

minimum, maximum = get_min_max([3, 1, 4, 1, 5])
```

## 4. 作用域

```python
x = 10  # 全局变量

def demo():
    y = 20      # 局部变量
    global x    # 声明使用全局变量
    x = 30

# nonlocal（嵌套函数中修改外层非全局变量）
def outer():
    count = 0
    def inner():
        nonlocal count
        count += 1
    inner()
    return count
```

## 5. Lambda 表达式

```python
# 简单匿名函数
square = lambda x: x ** 2
print(square(5))  # 25

# 与高阶函数结合
nums = [1, 2, 3, 4]
doubled = list(map(lambda x: x * 2, nums))  # [2, 4, 6, 8]
evens = list(filter(lambda x: x % 2 == 0, nums))  # [2, 4]

# 排序
students = [("Alice", 85), ("Bob", 92), ("Charlie", 78)]
students.sort(key=lambda x: x[1], reverse=True)
```

## 6. 递归

```python
# 阶乘
def factorial(n):
    if n <= 1:
        return 1
    return n * factorial(n - 1)

print(factorial(5))  # 120
```

## 笔记区

- 默认参数陷阱：不要用可变对象作为默认参数！
  ```python
  # ❌ 错误
  def append_item(item, lst=[]):
      lst.append(item)
      return lst
  
  # ✅ 正确
  def append_item(item, lst=None):
      if lst is None:
          lst = []
      lst.append(item)
      return lst
  ```
