# 08-常用标准库

## 1. os — 操作系统接口

```python
import os

os.getcwd()                 # 获取当前工作目录
os.chdir("/path/to/dir")    # 切换目录
os.listdir(".")             # 列出目录内容
os.mkdir("new_dir")         # 创建目录
os.remove("file.txt")       # 删除文件
os.rename("old", "new")     # 重命名

# 路径拼接（推荐）
path = os.path.join("folder", "subfolder", "file.txt")
```

## 2. datetime — 日期与时间

```python
from datetime import datetime, timedelta, date

now = datetime.now()
print(now.strftime("%Y-%m-%d %H:%M:%S"))  # 格式化输出

# 字符串转日期
birth = datetime.strptime("2000-05-20", "%Y-%m-%d")

# 时间运算
future = now + timedelta(days=7)
print(future)

# 日期差
delta = date(2024, 1, 1) - date.today()
print(delta.days)
```

## 3. json — JSON 数据处理

```python
import json

data = {"name": "Alice", "age": 25, "hobbies": ["reading", "coding"]}

# Python → JSON 字符串
json_str = json.dumps(data, ensure_ascii=False, indent=2)

# JSON 字符串 → Python
obj = json.loads(json_str)

# 读写 JSON 文件
with open("data.json", "w", encoding="utf-8") as f:
    json.dump(data, f, ensure_ascii=False, indent=2)

with open("data.json", "r", encoding="utf-8") as f:
    loaded = json.load(f)
```

## 4. re — 正则表达式

```python
import re

text = "我的邮箱是 alice@example.com，电话是 13800138000"

# 搜索
match = re.search(r"\w+@\w+\.\w+", text)
if match:
    print(match.group())  # alice@example.com

# 查找所有
emails = re.findall(r"\w+@\w+\.\w+", text)

# 替换
new_text = re.sub(r"\d{11}", "[电话号码隐藏]", text)

# 常用模式
# \d 数字, \w 单词字符, \s 空白, . 任意字符
# * 0次或多次, + 1次或多次, ? 0或1次, {n} n次
```

## 5. collections — 高级数据结构

```python
from collections import Counter, defaultdict, deque

# Counter: 计数器
words = ["apple", "banana", "apple", "cherry", "banana", "apple"]
count = Counter(words)
print(count.most_common(2))  # [('apple', 3), ('banana', 2)]

# defaultdict: 带默认值的字典
scores = defaultdict(int)
scores["Alice"] += 10   # 不会报错，默认 0

# deque: 双端队列（两端操作 O(1)）
d = deque([1, 2, 3])
d.appendleft(0)
d.pop()
```

## 6. itertools — 迭代器工具

```python
import itertools

# 无限计数器
for i in itertools.count(10, 2):  # 从10开始，步长2
    if i > 20:
        break
    print(i)

# 排列组合
list(itertools.permutations([1, 2, 3]))  # 排列
list(itertools.combinations([1, 2, 3], 2))  # 组合

# 拉链（不等长）
for a, b in itertools.zip_longest([1, 2], ["a", "b", "c"], fillvalue="-"):
    print(a, b)
```
